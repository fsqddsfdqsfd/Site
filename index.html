Résultat complet :

Section:CreateToggle("Mode invincible", function(state)
    local hum = game.Players.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.MaxHealth = state and math.huge or 100
        hum.Health    = state and math.huge or 100
    end
end)

Section:CreateSlider("Vitesse", 16, 120, function(val)
    local hum = game.Players.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed = val end
end)

Section:CreateDropdown("Choix de vitesse", {"16","50","100"}, function(choice)
    local hum = game.Players.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed = tonumber(choice) end
end)
