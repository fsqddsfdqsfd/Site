Bienvenue sur le tutoriel GitanX

GitanX est une librairie UI moderne pour Roblox. Ce site vous guide pas à pas pour l’utiliser en jeu.

Installation

loadstring(game:HttpGet("https://raw.githubusercontent.com/fsqddsfdqsfd/Site/refs/heads/main/test1.lua"))()

Résultat :

-- Ce script charge la librairie GitanX
-- Une fenêtre s’ouvre automatiquement avec le thème par défaut

Boutons

Section:CreateButton("Nom du bouton", function()
    -- votre script ici
end)

Résultat :

Section:CreateButton("Dire Bonjour", function()
    game:GetService("Players").LocalPlayer:Kick("Bonjour depuis GitanX !")
end)
-- Quand vous cliquez, le joueur est expulsé avec un message

Toggles

Section:CreateToggle("Nom du toggle", function(state)
    if state then
        -- votre script ici (activé)
    else
        -- votre script ici (désactivé)
    end
end)

Résultat :

Section:CreateToggle("Mode invisible", function(state)
    local char = game.Players.LocalPlayer.Character
    if char then
        char.Head.Transparency = state and 1 or 0
    end
end)
-- Active/désactive la transparence de la tête du joueur

Sliders

Section:CreateSlider("Nom du slider", 0, 100, function(val)
    -- votre script ici (val contient la valeur choisie)
end)

Résultat :

Section:CreateSlider("Taille du joueur", 1, 5, function(val)
    local char = game.Players.LocalPlayer.Character
    if char then
        char:SetPrimaryPartCFrame(char.PrimaryPart.CFrame * CFrame.new(0,0,0) * CFrame.Angles(0,0,0))
        char.PrimaryPart.Size = Vector3.new(val,val,val)
    end
end)
-- Permet de régler la taille du joueur entre 1 et 5

Dropdowns

Section:CreateDropdown("Nom du dropdown", {"Option1","Option2","Option3"}, function(choice)
    -- votre script ici (choice contient l'option choisie)
end)

Résultat :

Section:CreateDropdown("Choisir une couleur", {"Rouge","Vert","Bleu"}, function(choice)
    print("Couleur choisie:", choice)
end)
-- Affiche dans la console la couleur sélectionnée

Notifications

UI:Notify("Titre", "Message de notification")

Résultat :

UI:Notify("Info", "Bienvenue dans GitanX !")
-- Affiche une notification en jeu

Changer de thème

UI:ChangeTheme("Ocean") -- Thèmes disponibles: Default, Ocean, Amethyst, Light

Résultat :

UI:ChangeTheme("Amethyst")
-- Change le thème de l’interface en violet

FAQ

Q : Comment charger GitanX ?
R : Avec loadstring(game:HttpGet(...))().

Q : Quels contrôles sont disponibles ?
R : Boutons, Toggles, Sliders, Dropdowns, Notifications, Thèmes.

Q : Est-ce que ça marche en jeu ?
R : Oui, GitanX est conçu pour être exécuté directement dans Roblox.
